// Copy of Atlan4's "FPGA commands.txt" (D:\src\Fnirsi1013D\Verzia 1.xxx\FPGA for 1.xxx\), updated
// for fpga/al3/src/zaklad.v of fnirsi-1013d-mcp. Changes against Atlan4's file:
//  - 0x06 returns 0x15 0x3A (0x153A = 0x1539 with the ADC data captured through IOB registers at a per-core stage
//    (write 0x63, read 0x36 with outlier counters, write 0x64), B first by T/2 at every rate; 0x1539 = 0x1538 with the
//    read byte loaded on the falling edge of the MCU clock, the
//    start of the strobe, so it is settled on the pins before the MCU reads it; values, pointer advance and dummy strobe
//    unchanged; 0x1538 = 0x1537 with the encode timing fix and read 0x33; 0x1537 = 0x1536 with the
//    record-start and A/B-order fixes, no new command; 0x1536 = with
//    segments/averaging; 0x1535 = without segments/averaging, 0x1534 = decimation filter without the FIR,
//    0x1533 = EXT trigger only; Atlan4's 0x1532 AL3 design returns 0x15 0x32, his file still shows 0x14).
//  - 0x15 takes data[1:0]: 0 = CH1, 1 = CH2, 2 = EXT (calibration pin P133), 3 = reserved (acts as EXT).
//  - 0x28 (new read): external trigger status.
//  - 0x53, 0x54, 0x55 (new writes), 0x29 (new read): decimation filter (0x1534). With 0x53 = 0 (power-on)
//    the design behaves exactly as 0x1533.
//  - 0x56 (new write), 0x29 byte 0 bits 6/5: CIC compensation FIR (0x1535). With 0x56 = 0 (power-on) the
//    design behaves exactly as 0x1534.
//  - 0x57..0x5D (new writes), 0x2A..0x32 (new reads): segmented capture and hardware averaging (0x1536). With
//    0x57 = 0 (power-on) the design behaves exactly as 0x1535; 0x0A then reads as before.
//  - 0x33 (new read, 0x1538): encode self-check, 1 byte. Bit 0 sticky: a check failed since the last 0x01 = 1;
//    bit 1: the last write clock prediction was right; bits 3:2 = 01, encode convention 1 (B = write clock, A = its
//    complement, 1.25 ns later); bits 7:4 the last pin check {CH2 A, CH2 B, CH1 A, CH1 B}, 1 = right. A good arm
//    reads 0xF6. Write 0x33 (CH1 relays) is unchanged.
//  - 0x63, 0x64 (new writes), 0x36 (new read, 0x153A): ADC capture stage per core and the B-core reference edge
//    (0x63, 2 bytes; power-on 0x08E4 = CH1 A stage 4, CH1 B 4, CH2 A 3, CH2 B 4, B timed by its own encode), the
//    outlier threshold (0x64, power-on 8 codes) and the stages plus four 16-bit outlier counters (0x36, 10 bytes).
//    0x53 bits 3 / 4 read back but select nothing any more. Write 0x36 (CH2 relays) is unchanged. The legacy data
//    reach the memory 4 words later than on 0x1539 at 200 MSa/s (3 words at divided rates, 40 ns with K >= 1); EXT
//    is not delayed, so the EXT-to-data offset grows by that much (fpga/al3/README.md section 10).
//  - 0x0B, 0x0C, 0x1F high bytes: zaklad.v takes io_mcu_data[5:0] into [12:8] (bit 5 is dropped by the
//    13-bit register); this file showed [4:0].
// Multi-byte values are high byte first. Details: fpga/al3/README.md, section "Changes".

//================================ FPGA commands table ==================================
// Read timing (0x1539): every read byte below is loaded into o_mcu_data on the FALLING edge of i_mcu_clk (the start of
// the strobe; 0x1532..0x1538: the rising edge) and holds until the next strobe starts. The read pointer and mcu_state
// still advance on the rising edge, so the byte of a read strobe is the same as before; read the bus after raising the
// clock, as the firmware does.

//***************************************************************************************
//********************************** READING commands ***********************************
//***************************************************************************************

            8'h05: //Reading of the sampling system resets state
                    o_mcu_data[0] <= sample_system_state;
//---------------------------------------------------------------------------------------
            8'h06: //Reading of the FPGA version needs loading of high byte first
                  4'h0:
                    o_mcu_data <= 8'h15;
                  4'h1:
                    o_mcu_data <= 8'h3A;       //0x153A = AL3 with external trigger, decimation filter, FIR, segments, averaging,
                                               //         encode self-check, settled readout, staged ADC capture
            
            //When sampling finishes the writing is disabled. Enabled again with a reset.
            8'h0A: //Reading of the sampling done state
                    o_mcu_data[0] <= q_active ? q_done : ~sample_write_enable;
                    //0x1536: after an arm with 0x57 mode 1 or 2, bit0 = the whole run is done (all segments or
                    //all averages, or stopped); otherwise ~sample_write_enable as before

            8'h14: //Reading of the trigger address needs loading of high byte first
                   //Triggered: the word of the trigger (section below, +-1). Ended on the auto time-out or on 0x1A = 0
                   //written while the capture ran: 0x1532..0x153A pretrigger (record = ring 0 .. total - 1, spliced
                   //if the stop fell inside it); 0x153B W - total + pretrigger, W the address after the last word
                   //written, so 0x14 - pretrigger is the newest total words. Frozen with the write (0x153B).
                  4'h0:
                    o_mcu_data <= trigger_address[12:8];
                  4'h1:
                    o_mcu_data <= trigger_address[7:0];
            
            //This is not used in the software and is basically the same as command 0x0A
            //0x153B: no trigger is taken after the write has stopped, so bit0 describes the record (up to 0x153A an
            //edge after the stop set it)
            8'h18: //Reading of the sampling triggered state
                    o_mcu_data[0] <= sampling_triggered;
              
	    8'h1F //Loading two bytes of the data read address offset high byte first
                  4'h0:
		    sample_read_offset[12:8] <= io_mcu_data[5:0];
		  4'h1:
		    sample_read_offset[7:0] <= io_mcu_data;
            
            //0x20..0x23 read one byte of the 32-bit memory word and advance the read address by one word.
            //With K >= 1 (0x53) the word is { CH2[15:0], CH1[15:0] }: 0x20 = CH1 low byte, 0x21 = CH1 high byte,
            //0x22 = CH2 low byte, 0x23 = CH2 high byte.
            8'h20: //Reading of the sample data for channel 1 ADC 1 from memory
              o_mcu_data <= o_adc1A_d;
        
            8'h21: //Reading of the sample data for channel 1 ADC 2 from memory
              o_mcu_data <= o_adc1B_d;
          
            8'h22: //Reading of the sample data for channel 2 ADC 1 from memory
              o_mcu_data <= o_adc2A_d;
          
            8'h23: //Reading of the sample data for channel 2 ADC 2 from memory
              o_mcu_data <= o_adc2B_d;
           
            //0x24..0x27 (0x153A): the pin as the IOB register took it on the last rising clk_200MHz edge (a liveness
            //read; it is not one of the capture stages)
            8'h24: //Reading of the sample data for channel 1 ADC 1
              o_mcu_data <= adc_q1[7:0];
            
            8'h25: //Reading of the sample data for channel 1 ADC 2
              o_mcu_data <= adc_q1[15:8];
            
            8'h26: //Reading of the sample data for channel 2 ADC 1
              o_mcu_data <= adc_q1[23:16];
            
            8'h27: //Reading of the sample data for channel 2 ADC 2
              o_mcu_data <= adc_q1[31:24];

            8'h28: //Reading of the external trigger status (0x1533 and later)
                    o_mcu_data <= { 5'b0, ext_activity, gen_out_enable, ext_level };
                    //bit0 ext_level:      EXT pin level after the synchroniser and the majority filter
                    //bit1 gen_out_enable: 1 = the generator drives the pin (cmd 0x50), EXT then reads it back
                    //bit2 ext_activity:   an EXT edge (either direction) since the last sampling reset (cmd 0x01 = 1)

            8'h29: //Reading of the decimation geometry (0x1534 and later), 2 bytes
                  4'h0:
                    o_mcu_data <= { dec_on, fir_en, fir_act, dec_neg_b, dec_neg_a, dec_swap, dec_order };
                    //bit7 = 1: 16-bit samples, one per channel per memory word (K >= 1)
                    //bit6 = 0x56 bit 0 (FIR enable as written), bit5 = FIR active now: enable, K >= 1 and
                    //divider >= 14 with divider bit 1 set (N >= 16, N a multiple of 4); both 0 before 0x1535
                    //bit4/bit3 = B/A cores captured on the falling write clock edge, bit2 = core order swap,
                    //bits 1..0 = K (0x53)
                  4'h1:
                    o_mcu_data <= { 2'b0, dec_shift };   //output shift (0x54)

            8'h2A: //Sequencer status (0x1536), 8 bytes
                  4'h0:
                    o_mcu_data <= { q_done, q_ovf, q_running, q_fmt16, seq_raw, q_circ, q_mode };
                    //bit7 run done (0x0A), bit6 circular segments wrapped at least once, bit5 running,
                    //bit4 records are 16-bit (0x53 K >= 1 at the arm), bit3 raw readout (0x57 bit 3, live),
                    //bit2 circular, bits 1..0 mode of the run armed last (0 = none: 0x57 was 0 or 3)
                  4'h1..4'h3:
                    o_mcu_data <= q_cnt[23:16], q_cnt[15:8], q_cnt[7:0];
                    //segments completed, or records accumulated (all positions)
                  4'h4..4'h5:
                    o_mcu_data <= q_untrig[15:8], q_untrig[7:0];
                    //untriggered records (auto time-out): segments stored with triggered = 0, or averaging
                    //records skipped; saturates at 65535
                  4'h6..4'h7:
                    o_mcu_data <= { q_st[2:0], 1'b0, q_idx[11:8] }, q_idx[7:0];
                    //sequencer state (0 idle, 1 restart, 2 record, 3 post-pass, 4 table, 5 next, 6 hold-off,
                    //7 done) and the slot / position the next record goes to.
                    //The count can change between its bytes while running: read twice and compare.

            8'h2B..8'h32: //Accumulator / segment table word at the read pointer (0x1536), bytes 0..7
                    o_mcu_data <= seq_lat[8 * (cmd - 0x2B) +: 8];
                    //Same pointer and pipeline as 0x20..0x23: set it with 0x1F (bits 10..0 used), one dummy read,
                    //then each read advances the pointer by one word. View word:
                    //  segment mode, or 0x57 bit 3 raw: the stored 64-bit word
                    //    segment table entry s: [39:0] timestamp (5 ns units from the first write clock edge
                    //    after 0x01 = 0), [52:40] trigger offset in the slot (trigger_address mod 2^m),
                    //    [53] triggered, [63] valid
                    //    raw averaging, legacy: four 16-bit sums CH1A, CH1B, CH2A, CH2B; 16-bit: CH1 sum [31:0],
                    //    CH2 sum [63:32]
                    //  averaging, legacy: lanes [15:0] CH1A, [31:16] CH1B, [47:32] CH2A, [63:48] CH2B = sum >> S
                    //  averaging, 16-bit: [15:0] min(65535, CH1 sum >> S), [47:32] min(65535, CH2 sum >> S), rest 0

            8'h33: //Encode self-check (0x1538): { last pin check CH2A, CH2B, CH1A, CH1B, 2'b01, prediction ok, sticky error }
                    o_mcu_data <= { enc_ok, 2'b01, enc_pred_ok, enc_err };   //0xF6 when all is right

            8'h36: //ADC capture (0x153A), 10 bytes high byte first
                  4'h0, 4'h1:
                    o_mcu_data <= cap_sel[15:8], cap_sel[7:0];   //the 0x63 value in effect (power-on 0x08E4)
                  4'h2, 4'h3:
                    o_mcu_data <= oc_1A[15:8], oc_1A[7:0];       //CH1 A outlier count since the last 0x01 = 1
                  4'h4, 4'h5:
                    o_mcu_data <= oc_1B[15:8], oc_1B[7:0];       //CH1 B
                  4'h6, 4'h7:
                    o_mcu_data <= oc_2A[15:8], oc_2A[7:0];       //CH2 A
                  4'h8, 4'h9:
                    o_mcu_data <= oc_2B[15:8], oc_2B[7:0];       //CH2 B
                    //A counter is the number of memory words (from the third one after the release of 0x01) whose
                    //sample of that core differs from the core's previous word by more than the 0x64 threshold,
                    //counted while the capture writes (saturating at 65535). With a quiet input (noise under the
                    //threshold) a well-captured core reads 0; a core captured inside its data transition reads
                    //about every word. Read after the capture (0x0A = 1).
//---------------------------------------------------------------------------------------
            8'h41: //Reading of the touch panel address needs loading of high byte first
                  4'h0:
                    o_mcu_data <= touch_panel_address[15:8];
                  4'h1:
                    o_mcu_data <= touch_panel_address[7:0];

//***************************************************************************************
//********************************* WRITING commands ************************************
//***************************************************************************************
            
            8'h01: //Setting the sampling reset state
                    sample_system_reset <= io_mcu_data[0];
                    
            8'h0B: //Loading 2 bytes of pretrigger_samples (set_trigger_point)
                  4'h0:
                    pretrigger_samples[12:8] <= io_mcu_data[5:0];
                  4'h1:
                    pretrigger_samples[7:0] <= io_mcu_data;
            
            8'h0C: //Loading two bytes of the total number of sample high byte first
                  4'h0:
                    total_samples[12:8] <= io_mcu_data[5:0];
                  4'h1:
                    total_samples[7:0] <= io_mcu_data;
        
            8'h0D: //Loading of the four bytes of the sample rate divider high byte first
                  4'h0:
                    sample_rate_divider[31:24] <= io_mcu_data;
                  4'h1:
                    sample_rate_divider[23:16] <= io_mcu_data;
                  4'h2:

                  4'h3:
                    sample_rate_divider[7:0] <= io_mcu_data;

                   //0x0E + 2. 0x153B: it ends an untriggered capture only (a trigger before it completes its
                   //post-trigger words) and must be >= total - 2 for a record without words from before the reset.
                  4'h0:
                    time_base_set[31:24] <= io_mcu_data;
                  4'h1:

                  4'h2:

                    //EXT: calibration pin P133, 3.3 V logic, generator must be off (cmd 0x50 = 0).
                    //Edge from cmd 0x16, mode/pretrigger/total/time-out unchanged, level (0x17) unused.
                    //trigger_address (cmd 0x14) = last sample written before the EXT edge, as for CH1/CH2.
              
            8'h16: //Setting the trigger edge
                    trigger_edge <= io_mcu_data[0];
            
            8'h17: //Setting the trigger level
                    trigger_level <= io_mcu_data;
          
            8'h1A: //Setting the trigger mode
                    trigger_mode <= io_mcu_data[0];

                  4'h0:
                    hold_on_divider[31:24] <= io_mcu_data;

//---------------------------------------------------------------------------------------
// Decimation filter (0x1534). Details: fpga/al3/README.md "Changes" section 3.
// N = 2 * (floor(divider / 2) + 1), divider from 0x0D; output rate 200 MSa/s / N (odd divider: as K = 0).
// With K >= 1 the ADCs always run at 200 MSa/s and the filter output (gain N^K) is stored as
// min(65535, y >> shift). Exact while 255 * N^K < 2^48 (K = 3: N <= 10334).
// 0x0B, 0x0C, 0x0E and 0x14 count memory words = output samples. Latency from the captured word (adc*_w) to the
// memory: 12 write clocks; the capture itself adds its stages (0x153A, README section 10).

            8'h53: //Decimation filter configuration, 1 byte
                    dec_order <= io_mcu_data[1:0];  //K: 0 = sample skipping (0x1533), 1 = boxcar, 2 = CIC2, 3 = CIC3
                    dec_swap  <= io_mcu_data[2];    //0 = core B is the earlier sample of a pair (firmware interleave)
                    dec_neg_a <= io_mcu_data[3];    //0x1534..0x1539: A cores captured on the falling write clock edge;
                    dec_neg_b <= io_mcu_data[4];    //0x153A: read back in 0x29, no effect (0x63 selects the capture)

            8'h54: //Decimation filter output shift, 1 byte
                    dec_shift <= io_mcu_data[5:0];  //recommended max(0, bitlen(255 * N^K) - 16)

            8'h55: //Loading two bytes of the 16-bit trigger level high byte first (used when K >= 1, 0x17 is ignored)
                  4'h0:
                    trigger_level16[15:8] <= io_mcu_data;
                  4'h1:
                    trigger_level16[7:0] <= io_mcu_data;

//---------------------------------------------------------------------------------------
// CIC compensation FIR (0x1535). Details: fpga/al3/README.md "Changes" section 5.
// Engages when 0x56 bit 0 = 1, K >= 1, N >= 16 and N a multiple of 4 (divider >= 14, divider bit 1 set);
// otherwise the 0x1534 path runs unchanged. The CIC then decimates by N/2 (gain (N/2)^K), its output is
// min(65535, y >> shift), and a 15-tap symmetric FIR (DC gain 1, one tap set per K) decimates by 2:
// stored = clamp(0 .. 65535, round(FIR)). Output rate, memory layout and read-back as 0x1534.
// Recommended shift: max(0, bitlen(255 * (N/2)^K) - 16) + 1 (+1: headroom for the 6-9 % step overshoot).
// Latency pin to memory: 25 write clocks; EXT data lag 250 + 17.5 N + 1.25 K (N - 2) ns.
// After changing 0x0D, 0x53, 0x54 or 0x56 wait (K + 10) * N * 5 ns + 1 us before 0x01.

            8'h56: //CIC compensation FIR, 1 byte
                    fir_en <= io_mcu_data[0];       //1 = FIR on where eligible; bits 7..1 reserved, write 0

//---------------------------------------------------------------------------------------
// Segmented capture and hardware averaging (0x1536). Details: fpga/al3/README.md "Changes" section 6 and
// docs/plans/segmented-capture.md. Write 0x57..0x5C while 0x01 = 1 (they are latched there); 0x0B, 0x0C,
// 0x0E, 0x15..0x1A and 0x53..0x56 apply to every record of the run. A run ends with 0x0A = 1.
// Segments: 0x58 slots of 2^m words (0x59), slot s at words s * 2^m; per segment a normal capture with
// pretrigger 0x0B and total 0x0C (total <= 2^m - 2), then a table entry (read 0x2B..0x32 word s), the
// hold-off 0x5C and an automatic restart. Averaging: each triggered record (pretrigger included) is added
// into accumulator words p * total .. p * total + total - 1, p = record index mod 0x58 positions,
// 0x5A records per position; positions * total <= 2048, total <= 2046. Untriggered records are skipped.

            8'h57: //Sequencer mode, 1 byte
                    seq_mode <= io_mcu_data[1:0];   //0 off (power-on), 1 segments, 2 averaging, 3 off (reserved)
                    seq_circ <= io_mcu_data[2];     //segments: circular, wrap at 0x58 and run until 0x5D
                    seq_raw  <= io_mcu_data[3];     //averaging readout: raw sums (live, also after the run)
                                                    //bits 7..4 write 0

            8'h58: //Segments or averaging positions, 2 bytes high byte first
                  4'h0:
                    seq_count[11:8] <= io_mcu_data[3:0];
                  4'h1:
                    seq_count[7:0] <= io_mcu_data;  //segments 1 .. 8192 >> m, positions 1 .. 2048 / total

            8'h59: //Segment slot size, 1 byte
                    seg_m <= io_mcu_data[3:0];      //slot = 2^m words, m = 5 .. 12

            8'h5A: //Records per averaging position, 2 bytes high byte first
                  4'h0:
                    avg_n[15:8] <= io_mcu_data;
                  4'h1:
                    avg_n[7:0] <= io_mcu_data;      //1 .. 65535, 0 = 65536; legacy lanes clamp at 65535 above N = 257

            8'h5B: //Averaging readout shift, 1 byte
                    avg_shift <= io_mcu_data[4:0];  //S, live: the record can be read again with another S

            8'h5C: //Hold-off, 4 bytes high byte first, 5 ns units
                  4'h0..4'h3:
                    seq_hold[31:24], [23:16], [15:8], [7:0] <= io_mcu_data;
                    //wait from the end of a segment's table write / a record's post-pass to the restart

            8'h5D: //Stop, 1 byte
                    seq_stop <= io_mcu_data[0];     //1 = no further restarts: an untriggered record in progress is
                                                    //abandoned (its slot entry stays invalid), a triggered one
                                                    //completes; then 0x0A = 1. Cleared by 0x01 = 1.

//---------------------------------------------------------------------------------------
// ADC capture (0x153A). Every ADC pin is sampled by an IOB DDR register on clk_200MHz; a write clock edge E sees
// the pins as they were 5 + 2.5 k ns before it (stage k = 0..5). Core A of a chip is encoded on the falling write
// clock edge and captured against it, core B on the rising edge and captured against that, so each core's data
// window has the same position relative to its own edge at every sample rate; stages 3 and 4 lie inside it with
// 1.3..2 ns to spare against the AD9288 limits (tV 2 ns, tPD 6 ns) on the owner's unit per TD's numbers, 2 and 5
// are reserves for a drifted window. A word holds the B sample encoded T/2 before its A sample (B first) at every
// rate. Setting a core inside its transition gives rail values and jumps, not a shifted trace: train with 0x36.

            8'h63: //ADC capture stages and B reference, 2 bytes high byte first; power-on 0x08E4
                  4'h0:
                    cap_sel[15:8] <= io_mcu_data;   //bit 13 CH2 B / bit 12 CH1 B timed by the chip's A encode (ADC
                                                    //data-align mode; power-on 0 = by its own encode), [11:9] CH2 B stage
                  4'h1:
                    cap_sel[7:0] <= io_mcu_data;    //[8:6] CH2 A stage, [5:3] CH1 B, [2:0] CH1 A (6 and 7 act as 5)
                    //Takes effect at the next write clock edges; write it before 0x01 = 1 of the capture it is for.

            8'h64: //Outlier threshold for the 0x36 counters, 1 byte, codes; power-on 8
                    cap_thr <= io_mcu_data;
